//! `fogu:s3` module — S3-compatible object storage (AWS S3, MinIO,
//! Cloudflare R2, Backblaze B2, DigitalOcean Spaces, Wasabi…), with AWS
//! Signature V4 signing.
//!
//! ```fogu
//! import s3 from "fogu:s3"
//!
//! const store = s3.connect({ bucket: "assets", region: "us-east-1" })?   // $AWS_ACCESS_KEY_ID…
//! // MinIO / R2: { endpoint: "http://127.0.0.1:9000", accessKey, secretKey, bucket }
//! store.put("notes/hello.txt", "Hello!", { contentType: "text/plain" })?
//! print(store.get("notes/hello.txt")?)
//! for obj in store.list("notes/")? { print(obj.key, obj.size) }
//! print(store.presign("notes/hello.txt", { expires: 15m }))           // shareable URL
//! store.delete("notes/hello.txt")?
//! ```
//!
//! Requests run on a background thread and wait cooperatively. Network and
//! storage failures (missing key, access denied) are `error` values.

use std::cell::RefCell;
use std::collections::{BTreeMap, HashMap};
use std::time::{Duration, SystemTime, UNIX_EPOCH};

use hmac::{Hmac, Mac};
use sha2::{Digest, Sha256};

use crate::runtime::value::{ModuleValue, Value};

use super::object::{FoguObject, NativeObjectRef, Poller};
use super::util::{duration_arg, list, map, max_args, nfn_var, str_arg, thread_poller};

pub fn build() -> ModuleValue {
    let mut e: HashMap<String, Value> = HashMap::new();
    e.insert("connect".into(), nfn_var("connect", 0, connect));
    ModuleValue {
        name: "fogu:s3".into(),
        exports: RefCell::new(e),
    }
}

#[derive(Clone, Debug)]
struct Cfg {
    /// `https://s3.us-east-1.amazonaws.com` or a custom endpoint
    endpoint: String,
    custom_endpoint: bool,
    region: String,
    access_key: String,
    secret_key: String,
    session_token: Option<String>,
    bucket: Option<String>,
    path_style: bool,
    timeout: Duration,
}

fn get_s(m: &HashMap<String, Value>, k: &str) -> Option<String> {
    match m.get(k) {
        Some(Value::Str(s)) if !s.is_empty() => Some(s.clone()),
        _ => None,
    }
}

fn env(k: &str) -> Option<String> {
    std::env::var(k).ok().filter(|v| !v.is_empty())
}

/// `connect({ endpoint, region, accessKey, secretKey, sessionToken, bucket, pathStyle, timeout })`.
/// Missing credentials come from `AWS_ACCESS_KEY_ID`, `AWS_SECRET_ACCESS_KEY`,
/// `AWS_SESSION_TOKEN`, `AWS_REGION` and `AWS_ENDPOINT_URL`.
fn parse_cfg(arg: Option<&Value>) -> Result<Cfg, String> {
    let m = match arg {
        None | Some(Value::Nil) => HashMap::new(),
        Some(Value::Map(m)) => m.borrow().clone(),
        Some(other) => return Err(format!("s3.connect expects an options map, got '{}'", other.type_name())),
    };
    let region = get_s(&m, "region")
        .or_else(|| env("AWS_REGION"))
        .or_else(|| env("AWS_DEFAULT_REGION"))
        .unwrap_or_else(|| "us-east-1".into());
    let endpoint = get_s(&m, "endpoint").or_else(|| env("AWS_ENDPOINT_URL_S3")).or_else(|| env("AWS_ENDPOINT_URL"));
    let custom = endpoint.is_some();
    let endpoint = endpoint.unwrap_or_else(|| format!("https://s3.{}.amazonaws.com", region));
    if !endpoint.starts_with("http://") && !endpoint.starts_with("https://") {
        return Err(format!("s3.connect: endpoint must start with http:// or https://, got '{}'", endpoint));
    }
    let access_key = get_s(&m, "accessKey")
        .or_else(|| get_s(&m, "accessKeyId"))
        .or_else(|| env("AWS_ACCESS_KEY_ID"))
        .ok_or("s3.connect: needs 'accessKey' (or $AWS_ACCESS_KEY_ID)")?;
    let secret_key = get_s(&m, "secretKey")
        .or_else(|| get_s(&m, "secretAccessKey"))
        .or_else(|| env("AWS_SECRET_ACCESS_KEY"))
        .ok_or("s3.connect: needs 'secretKey' (or $AWS_SECRET_ACCESS_KEY)")?;
    let path_style = match m.get("pathStyle") {
        Some(Value::Bool(b)) => *b,
        _ => custom,
    };
    Ok(Cfg {
        endpoint: endpoint.trim_end_matches('/').to_string(),
        custom_endpoint: custom,
        region,
        access_key,
        secret_key,
        session_token: get_s(&m, "sessionToken").or_else(|| env("AWS_SESSION_TOKEN")),
        bucket: get_s(&m, "bucket"),
        path_style,
        timeout: match m.get("timeout") {
            None | Some(Value::Nil) => Duration::from_secs(60),
            Some(t) => duration_arg(t, "s3.connect: timeout")?,
        },
    })
}

// Signature V4
// ---------------------------------------------------------------------

fn hex(b: &[u8]) -> String {
    b.iter().map(|x| format!("{:02x}", x)).collect()
}

fn sha256_hex(data: &[u8]) -> String {
    hex(&Sha256::digest(data))
}

fn hmac(key: &[u8], data: &str) -> Vec<u8> {
    let mut m = <Hmac<Sha256>>::new_from_slice(key).expect("HMAC accepts any key length");
    m.update(data.as_bytes());
    m.finalize().into_bytes().to_vec()
}

/// RFC 3986 encoding; `/` is kept in paths.
fn uri_encode(s: &str, keep_slash: bool) -> String {
    let mut out = String::with_capacity(s.len());
    for b in s.bytes() {
        match b {
            b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'_' | b'.' | b'~' => out.push(b as char),
            b'/' if keep_slash => out.push('/'),
            _ => out.push_str(&format!("%{:02X}", b)),
        }
    }
    out
}

/// `(YYYYMMDD'T'HHMMSS'Z', YYYYMMDD)` for a UNIX time.
fn amz_dates(secs: u64) -> (String, String) {
    let days = (secs / 86_400) as i64;
    let rem = secs % 86_400;
    let z = days + 719_468;
    let era = z.div_euclid(146_097);
    let doe = z - era * 146_097;
    let yoe = (doe - doe / 1460 + doe / 36_524 - doe / 146_096) / 365;
    let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
    let mp = (5 * doy + 2) / 153;
    let d = doy - (153 * mp + 2) / 5 + 1;
    let m = if mp < 10 { mp + 3 } else { mp - 9 };
    let y = yoe + era * 400 + i64::from(m <= 2);
    let date = format!("{:04}{:02}{:02}", y, m, d);
    (format!("{}T{:02}{:02}{:02}Z", date, rem / 3600, rem % 3600 / 60, rem % 60), date)
}

fn now_secs() -> u64 {
    SystemTime::now().duration_since(UNIX_EPOCH).map(|d| d.as_secs()).unwrap_or(0)
}

struct Target {
    /// scheme://host[:port]
    origin: String,
    /// host[:port] as sent in the Host header
    host: String,
    /// encoded path
    path: String,
}

fn target(cfg: &Cfg, bucket: Option<&str>, key: Option<&str>) -> Target {
    let (scheme, rest) = cfg.endpoint.split_once("://").unwrap();
    let (host, base_path) = match rest.find('/') {
        Some(i) => (&rest[..i], &rest[i..]),
        None => (rest, ""),
    };
    let key_path = key.map(|k| uri_encode(k, true)).unwrap_or_default();
    match bucket {
        Some(b) if !cfg.path_style => {
            let host = format!("{}.{}", b, host);
            Target {
                origin: format!("{}://{}", scheme, host),
                host,
                path: format!("{}/{}", base_path, key_path),
            }
        }
        Some(b) => Target {
            origin: format!("{}://{}", scheme, host),
            host: host.to_string(),
            path: format!("{}/{}/{}", base_path, uri_encode(b, false), key_path)
                .trim_end_matches(|c| c == '/' && key.is_none())
                .to_string()
                .replace("//", "/"),
        },
        None => Target { origin: format!("{}://{}", scheme, host), host: host.to_string(), path: format!("{}/", base_path) },
    }
}

fn canonical_query(q: &BTreeMap<String, String>) -> String {
    q.iter().map(|(k, v)| format!("{}={}", uri_encode(k, false), uri_encode(v, false))).collect::<Vec<_>>().join("&")
}

fn signing_key(cfg: &Cfg, date: &str) -> Vec<u8> {
    let k = hmac(format!("AWS4{}", cfg.secret_key).as_bytes(), date);
    let k = hmac(&k, &cfg.region);
    let k = hmac(&k, "s3");
    hmac(&k, "aws4_request")
}

/// Signed headers for a request (the `Authorization` header included).
fn sign(
    cfg: &Cfg,
    method: &str,
    t: &Target,
    query: &BTreeMap<String, String>,
    extra: &BTreeMap<String, String>,
    payload_hash: &str,
    now: u64,
) -> BTreeMap<String, String> {
    let (amz_date, date) = amz_dates(now);
    let mut headers: BTreeMap<String, String> = extra.iter().map(|(k, v)| (k.to_lowercase(), v.trim().to_string())).collect();
    headers.insert("host".into(), t.host.clone());
    headers.insert("x-amz-date".into(), amz_date.clone());
    headers.insert("x-amz-content-sha256".into(), payload_hash.into());
    if let Some(tok) = &cfg.session_token {
        headers.insert("x-amz-security-token".into(), tok.clone());
    }
    let signed: Vec<&str> = headers.keys().map(String::as_str).collect();
    let canonical = format!(
        "{}\n{}\n{}\n{}\n{}\n{}",
        method,
        t.path,
        canonical_query(query),
        headers.iter().map(|(k, v)| format!("{}:{}\n", k, v)).collect::<String>(),
        signed.join(";"),
        payload_hash
    );
    let scope = format!("{}/{}/s3/aws4_request", date, cfg.region);
    let to_sign = format!("AWS4-HMAC-SHA256\n{}\n{}\n{}", amz_date, scope, sha256_hex(canonical.as_bytes()));
    let signature = hex(&hmac(&signing_key(cfg, &date), &to_sign));
    let auth = format!(
        "AWS4-HMAC-SHA256 Credential={}/{}, SignedHeaders={}, Signature={}",
        cfg.access_key,
        scope,
        signed.join(";"),
        signature
    );
    headers.insert("authorization".into(), auth);
    headers
}

/// A pre-signed URL (query string authentication).
fn presign(cfg: &Cfg, method: &str, bucket: &str, key: &str, expires: u64, now: u64) -> String {
    let t = target(cfg, Some(bucket), Some(key));
    let (amz_date, date) = amz_dates(now);
    let scope = format!("{}/{}/s3/aws4_request", date, cfg.region);
    let mut q: BTreeMap<String, String> = BTreeMap::new();
    q.insert("X-Amz-Algorithm".into(), "AWS4-HMAC-SHA256".into());
    q.insert("X-Amz-Credential".into(), format!("{}/{}", cfg.access_key, scope));
    q.insert("X-Amz-Date".into(), amz_date.clone());
    q.insert("X-Amz-Expires".into(), expires.to_string());
    q.insert("X-Amz-SignedHeaders".into(), "host".into());
    if let Some(tok) = &cfg.session_token {
        q.insert("X-Amz-Security-Token".into(), tok.clone());
    }
    let canonical = format!("{}\n{}\n{}\nhost:{}\n\nhost\nUNSIGNED-PAYLOAD", method, t.path, canonical_query(&q), t.host);
    let to_sign = format!("AWS4-HMAC-SHA256\n{}\n{}\n{}", amz_date, scope, sha256_hex(canonical.as_bytes()));
    let signature = hex(&hmac(&signing_key(cfg, &date), &to_sign));
    format!("{}{}?{}&X-Amz-Signature={}", t.origin, t.path, canonical_query(&q), signature)
}

struct Resp {
    status: u16,
    headers: reqwest::header::HeaderMap,
    body: Vec<u8>,
}

#[allow(clippy::too_many_arguments)]
fn request(
    cfg: &Cfg,
    method: &str,
    bucket: Option<&str>,
    key: Option<&str>,
    query: BTreeMap<String, String>,
    extra: BTreeMap<String, String>,
    body: Vec<u8>,
) -> Result<Resp, String> {
    let t = target(cfg, bucket, key);
    let headers = sign(cfg, method, &t, &query, &extra, &sha256_hex(&body), now_secs());
    let qs = canonical_query(&query);
    let url = if qs.is_empty() { format!("{}{}", t.origin, t.path) } else { format!("{}{}?{}", t.origin, t.path, qs) };
    let client = reqwest::blocking::Client::builder()
        .timeout(cfg.timeout)
        .build()
        .map_err(|e| io_err!("s3: {}", e))?;
    let mut rb = client.request(reqwest::Method::from_bytes(method.as_bytes()).unwrap(), url);
    for (k, v) in &headers {
        if k != "host" {
            rb = rb.header(k.as_str(), v.as_str());
        }
    }
    let resp = rb.body(body).send().map_err(|e| io_err!("s3: {}", e))?;
    let status = resp.status().as_u16();
    let headers = resp.headers().clone();
    let body = resp.bytes().map_err(|e| io_err!("s3: {}", e))?.to_vec();
    Ok(Resp { status, headers, body })
}

fn xml_unescape(s: &str) -> String {
    s.replace("&lt;", "<").replace("&gt;", ">").replace("&quot;", "\"").replace("&apos;", "'").replace("&#39;", "'").replace("&amp;", "&")
}

fn tag(xml: &str, name: &str) -> Option<String> {
    let open = format!("<{}>", name);
    let close = format!("</{}>", name);
    let start = xml.find(&open)? + open.len();
    let end = xml[start..].find(&close)? + start;
    Some(xml_unescape(&xml[start..end]))
}

fn blocks<'a>(xml: &'a str, name: &str) -> Vec<&'a str> {
    let open = format!("<{}>", name);
    let close = format!("</{}>", name);
    let mut out = Vec::new();
    let mut rest = xml;
    while let Some(s) = rest.find(&open) {
        let after = &rest[s + open.len()..];
        match after.find(&close) {
            Some(e) => {
                out.push(&after[..e]);
                rest = &after[e + close.len()..];
            }
            None => break,
        }
    }
    out
}

/// `Code: Message` from an error answer.
fn s3_error(what: &str, r: &Resp) -> String {
    let xml = String::from_utf8_lossy(&r.body);
    let detail = match (tag(&xml, "Code"), tag(&xml, "Message")) {
        (Some(c), Some(m)) => format!("{}: {}", c, m),
        (Some(c), None) => c,
        _ => match r.status {
            403 => "AccessDenied".into(),
            404 => "NotFound".into(),
            s => format!("HTTP {}", s),
        },
    };
    io_err!("s3: {}: {}", what, detail)
}

fn ok_or(what: &str, r: Resp) -> Result<Resp, String> {
    if (200..300).contains(&r.status) { Ok(r) } else { Err(s3_error(what, &r)) }
}

fn header(r: &Resp, k: &str) -> Option<String> {
    r.headers.get(k).and_then(|v| v.to_str().ok()).map(String::from)
}

#[derive(Debug, Default)]
struct Obj {
    key: String,
    size: i64,
    etag: String,
    last_modified: String,
    dir: bool,
}

/// ListObjectsV2, following continuation tokens up to `max` objects.
fn list_objects(cfg: &Cfg, bucket: &str, prefix: &str, delimiter: Option<&str>, max: Option<usize>) -> Result<Vec<Obj>, String> {
    let mut out = Vec::new();
    let mut token: Option<String> = None;
    loop {
        let mut q = BTreeMap::new();
        q.insert("list-type".to_string(), "2".to_string());
        if !prefix.is_empty() {
            q.insert("prefix".into(), prefix.into());
        }
        if let Some(d) = delimiter {
            q.insert("delimiter".into(), d.into());
        }
        if let Some(t) = &token {
            q.insert("continuation-token".into(), t.clone());
        }
        if let Some(m) = max {
            q.insert("max-keys".into(), (m - out.len()).min(1000).to_string());
        }
        let r = ok_or(&format!("list {}", bucket), request(cfg, "GET", Some(bucket), None, q, BTreeMap::new(), Vec::new())?)?;
        let xml = String::from_utf8_lossy(&r.body).to_string();
        for b in blocks(&xml, "CommonPrefixes") {
            out.push(Obj { key: tag(b, "Prefix").unwrap_or_default(), dir: true, ..Default::default() });
        }
        for b in blocks(&xml, "Contents") {
            out.push(Obj {
                key: tag(b, "Key").unwrap_or_default(),
                size: tag(b, "Size").and_then(|s| s.parse().ok()).unwrap_or(0),
                etag: tag(b, "ETag").unwrap_or_default().trim_matches('"').to_string(),
                last_modified: tag(b, "LastModified").unwrap_or_default(),
                dir: false,
            });
        }
        let truncated = tag(&xml, "IsTruncated").as_deref() == Some("true");
        token = tag(&xml, "NextContinuationToken");
        if !truncated || token.is_none() || max.is_some_and(|m| out.len() >= m) {
            break;
        }
    }
    out.sort_by(|a, b| a.key.cmp(&b.key));
    Ok(out)
}

fn connect(args: &[Value]) -> Result<Value, String> {
    max_args(args, 1, "s3.connect")?;
    let cfg = parse_cfg(args.first())?;
    Ok(NativeObjectRef::new(S3Value { cfg }).to_value())
}

pub struct S3Value {
    cfg: Cfg,
}

/// Bytes to upload: a str as is, a list of byte ints (0..255) as binary,
/// anything else as JSON. Files go through `upload(path, key)`.
fn body_of(v: &Value) -> Vec<u8> {
    match v {
        Value::Str(s) => s.as_bytes().to_vec(),
        Value::List(l) => {
            let items = l.borrow();
            let bytes: Option<Vec<u8>> = items
                .iter()
                .map(|b| match b {
                    Value::Int(n) if (0..=255).contains(n) => Some(*n as u8),
                    _ => None,
                })
                .collect();
            bytes.unwrap_or_else(|| super::json::stringify_value(v).into_bytes())
        }
        Value::Nil => Vec::new(),
        other => super::json::stringify_value(other).into_bytes(),
    }
}

fn guess_type(key: &str) -> &'static str {
    match key.rsplit('.').next().unwrap_or("").to_lowercase().as_str() {
        "txt" | "log" | "md" => "text/plain; charset=utf-8",
        "html" | "htm" => "text/html; charset=utf-8",
        "css" => "text/css",
        "js" | "mjs" => "text/javascript",
        "json" => "application/json",
        "csv" => "text/csv",
        "xml" => "application/xml",
        "png" => "image/png",
        "jpg" | "jpeg" => "image/jpeg",
        "gif" => "image/gif",
        "svg" => "image/svg+xml",
        "webp" => "image/webp",
        "pdf" => "application/pdf",
        "zip" => "application/zip",
        "wasm" => "application/wasm",
        "mp4" => "video/mp4",
        _ => "application/octet-stream",
    }
}

impl S3Value {
    fn bucket(&self, o: &HashMap<String, Value>, what: &str) -> Result<String, String> {
        get_s(o, "bucket")
            .or_else(|| self.cfg.bucket.clone())
            .ok_or_else(|| format!("{}: no bucket (pass {{ bucket }} or connect with one)", what))
    }
}

fn opts(args: &[Value], i: usize, what: &str) -> Result<HashMap<String, Value>, String> {
    match args.get(i) {
        None | Some(Value::Nil) => Ok(HashMap::new()),
        Some(Value::Map(m)) => Ok(m.borrow().clone()),
        Some(other) => Err(format!("{}: options must be a map, got '{}'", what, other.type_name())),
    }
}

fn obj_value(o: &Obj) -> Value {
    if o.dir {
        return map(vec![("key", Value::Str(o.key.clone())), ("dir", Value::Bool(true))]);
    }
    map(vec![
        ("key", Value::Str(o.key.clone())),
        ("size", Value::Int(o.size)),
        ("etag", Value::Str(o.etag.clone())),
        ("lastModified", Value::Str(o.last_modified.clone())),
    ])
}

impl FoguObject for S3Value {
    fn name(&self) -> &str {
        "S3"
    }

    fn get(&self, field: &str) -> Option<Value> {
        Some(match field {
            "bucket" => self.cfg.bucket.clone().map_or(Value::Nil, Value::Str),
            "region" => Value::Str(self.cfg.region.clone()),
            "endpoint" => Value::Str(self.cfg.endpoint.clone()),
            _ => return None,
        })
    }

    fn call(&mut self, method: &str, args: &[Value]) -> Result<Value, String> {
        match method {
            // withBucket(name) → a client bound to another bucket
            "withBucket" => {
                max_args(args, 1, "withBucket")?;
                let mut cfg = self.cfg.clone();
                cfg.bucket = Some(str_arg(args, 0, "withBucket")?);
                Ok(NativeObjectRef::new(S3Value { cfg }).to_value())
            }
            // presign(key, { method = "GET", expires = 1h, bucket }) → URL
            "presign" => {
                max_args(args, 2, "presign")?;
                let key = str_arg(args, 0, "presign")?;
                let o = opts(args, 1, "presign")?;
                let bucket = self.bucket(&o, "presign")?;
                let method = get_s(&o, "method").unwrap_or_else(|| "GET".into()).to_uppercase();
                let expires = match o.get("expires") {
                    None | Some(Value::Nil) => 3600,
                    Some(v) => duration_arg(v, "presign: expires")?.as_secs(),
                };
                if !(1..=604_800).contains(&expires) {
                    return Err("presign: expires must be between 1s and 7 days".into());
                }
                Ok(Value::Str(presign(&self.cfg, &method, &bucket, &key, expires, now_secs())))
            }
            // url(key) → the plain (unsigned) URL of an object
            "url" => {
                max_args(args, 2, "url")?;
                let key = str_arg(args, 0, "url")?;
                let bucket = self.bucket(&opts(args, 1, "url")?, "url")?;
                let t = target(&self.cfg, Some(&bucket), Some(&key));
                Ok(Value::Str(format!("{}{}", t.origin, t.path)))
            }
            other => Err(format!(
                "method '{}' does not exist on S3 (put get getBytes download upload head exists delete list copy presign url withBucket createBucket deleteBucket listBuckets)",
                other
            )),
        }
    }

    fn start_async(&mut self, method: &str, args: &[Value]) -> Option<Result<Poller, String>> {
        let cfg = self.cfg.clone();
        let r = (|| -> Result<Option<Poller>, String> {
            Ok(Some(match method {
                // put(key, body, { contentType, metadata, cacheControl, acl, bucket }) → { etag }
                "put" | "upload" => {
                    max_args(args, 3, method)?;
                    let (key, body, o) = if method == "upload" {
                        // upload(path, key?, opts?)
                        let path = str_arg(args, 0, "upload")?;
                        let key = match args.get(1) {
                            Some(Value::Str(k)) => k.clone(),
                            _ => path.replace('\\', "/").rsplit('/').next().unwrap_or(&path).to_string(),
                        };
                        let body = std::fs::read(&path).map_err(|e| io_err!("s3.upload: cannot read {}: {}", path, e))?;
                        (key, body, opts(args, 2, "upload")?)
                    } else {
                        let key = str_arg(args, 0, "put")?;
                        let body = body_of(args.get(1).unwrap_or(&Value::Nil));
                        (key, body, opts(args, 2, "put")?)
                    };
                    let bucket = self.bucket(&o, method)?;
                    let mut h = BTreeMap::new();
                    h.insert("content-type".to_string(), get_s(&o, "contentType").unwrap_or_else(|| guess_type(&key).into()));
                    if let Some(c) = get_s(&o, "cacheControl") {
                        h.insert("cache-control".into(), c);
                    }
                    if let Some(a) = get_s(&o, "acl") {
                        h.insert("x-amz-acl".into(), a);
                    }
                    if let Some(Value::Map(meta)) = o.get("metadata") {
                        for (k, v) in meta.borrow().iter() {
                            h.insert(format!("x-amz-meta-{}", k.to_lowercase()), v.to_display_string());
                        }
                    }
                    thread_poller(
                        move || {
                            let r = ok_or(&format!("put {}", key), request(&cfg, "PUT", Some(&bucket), Some(&key), BTreeMap::new(), h, body)?)?;
                            Ok(header(&r, "etag").unwrap_or_default().trim_matches('"').to_string())
                        },
                        |etag| Ok(map(vec![("etag", Value::Str(etag))])),
                    )
                }
                // get(key) → str · getBytes(key) → [int] · download(key, path)
                "get" | "getBytes" | "download" => {
                    max_args(args, if method == "download" { 3 } else { 2 }, method)?;
                    let key = str_arg(args, 0, method)?;
                    let path = if method == "download" { Some(str_arg(args, 1, "download")?) } else { None };
                    let o = opts(args, if method == "download" { 2 } else { 1 }, method)?;
                    let bucket = self.bucket(&o, method)?;
                    let kind = method.to_string();
                    thread_poller(
                        move || {
                            let r = ok_or(&format!("get {}", key), request(&cfg, "GET", Some(&bucket), Some(&key), BTreeMap::new(), BTreeMap::new(), Vec::new())?)?;
                            if let Some(p) = &path {
                                if let Some(dir) = std::path::Path::new(p).parent().filter(|d| !d.as_os_str().is_empty()) {
                                    std::fs::create_dir_all(dir).map_err(|e| io_err!("s3.download: {}: {}", p, e))?;
                                }
                                std::fs::write(p, &r.body).map_err(|e| io_err!("s3.download: {}: {}", p, e))?;
                            }
                            Ok(r.body)
                        },
                        move |body| {
                            Ok(match kind.as_str() {
                                "get" => Value::Str(String::from_utf8_lossy(&body).into_owned()),
                                "getBytes" => list(body.into_iter().map(|b| Value::Int(b as i64)).collect()),
                                _ => Value::Nil,
                            })
                        },
                    )
                }
                // head(key) → { size, etag, contentType, lastModified, metadata } or nil · exists(key) → bool
                "head" | "exists" => {
                    max_args(args, 2, method)?;
                    let key = str_arg(args, 0, method)?;
                    let bucket = self.bucket(&opts(args, 1, method)?, method)?;
                    let exists = method == "exists";
                    thread_poller(
                        move || {
                            let r = request(&cfg, "HEAD", Some(&bucket), Some(&key), BTreeMap::new(), BTreeMap::new(), Vec::new())?;
                            if r.status == 404 {
                                return Ok(None);
                            }
                            let r = ok_or(&format!("head {}", key), r)?;
                            let meta: Vec<(String, String)> = r
                                .headers
                                .iter()
                                .filter_map(|(k, v)| {
                                    k.as_str().strip_prefix("x-amz-meta-").map(|n| (n.to_string(), v.to_str().unwrap_or("").to_string()))
                                })
                                .collect();
                            Ok(Some((
                                header(&r, "content-length").and_then(|s| s.parse::<i64>().ok()).unwrap_or(0),
                                header(&r, "etag").unwrap_or_default().trim_matches('"').to_string(),
                                header(&r, "content-type").unwrap_or_default(),
                                header(&r, "last-modified").unwrap_or_default(),
                                meta,
                            )))
                        },
                        move |found| {
                            if exists {
                                return Ok(Value::Bool(found.is_some()));
                            }
                            Ok(match found {
                                None => Value::Nil,
                                Some((size, etag, ct, lm, meta)) => map(vec![
                                    ("size", Value::Int(size)),
                                    ("etag", Value::Str(etag)),
                                    ("contentType", Value::Str(ct)),
                                    ("lastModified", Value::Str(lm)),
                                    ("metadata", map(meta.iter().map(|(k, v)| (k.as_str(), Value::Str(v.clone()))).collect())),
                                ]),
                            })
                        },
                    )
                }
                "delete" => {
                    max_args(args, 2, "delete")?;
                    let keys: Vec<String> = match args.first() {
                        Some(Value::Str(k)) => vec![k.clone()],
                        Some(Value::List(l)) => l.borrow().iter().map(|v| v.to_display_string()).collect(),
                        _ => return Err("delete expects a key or a list of keys".into()),
                    };
                    let bucket = self.bucket(&opts(args, 1, "delete")?, "delete")?;
                    thread_poller(
                        move || {
                            for k in &keys {
                                ok_or(&format!("delete {}", k), request(&cfg, "DELETE", Some(&bucket), Some(k), BTreeMap::new(), BTreeMap::new(), Vec::new())?)?;
                            }
                            Ok(())
                        },
                        |_| Ok(Value::Nil),
                    )
                }
                // list(prefix?, { delimiter, max, bucket }) → [{ key, size, etag, lastModified } | { key, dir }]
                "list" => {
                    max_args(args, 2, "list")?;
                    let prefix = match args.first() {
                        Some(Value::Str(p)) => p.clone(),
                        _ => String::new(),
                    };
                    let o = opts(args, 1, "list")?;
                    let bucket = self.bucket(&o, "list")?;
                    let delimiter = get_s(&o, "delimiter");
                    let max = match o.get("max") {
                        Some(Value::Int(n)) if *n > 0 => Some(*n as usize),
                        _ => None,
                    };
                    thread_poller(
                        move || list_objects(&cfg, &bucket, &prefix, delimiter.as_deref(), max),
                        |objs| Ok(list(objs.iter().map(obj_value).collect())),
                    )
                }
                // copy(from, to, { fromBucket, bucket })
                "copy" => {
                    max_args(args, 3, "copy")?;
                    let from = str_arg(args, 0, "copy")?;
                    let to = str_arg(args, 1, "copy")?;
                    let o = opts(args, 2, "copy")?;
                    let bucket = self.bucket(&o, "copy")?;
                    let from_bucket = get_s(&o, "fromBucket").unwrap_or_else(|| bucket.clone());
                    let mut h = BTreeMap::new();
                    h.insert("x-amz-copy-source".to_string(), format!("/{}/{}", from_bucket, uri_encode(&from, true)));
                    thread_poller(
                        move || {
                            let r = request(&cfg, "PUT", Some(&bucket), Some(&to), BTreeMap::new(), h, Vec::new())?;
                            let r = ok_or(&format!("copy {} → {}", from, to), r)?;
                            // S3 may answer 200 with an error document
                            let xml = String::from_utf8_lossy(&r.body).to_string();
                            if xml.contains("<Error>") {
                                return Err(s3_error(&format!("copy {}", from), &r));
                            }
                            Ok(tag(&xml, "ETag").unwrap_or_default().trim_matches('"').to_string())
                        },
                        |etag| Ok(map(vec![("etag", Value::Str(etag))])),
                    )
                }
                "createBucket" | "deleteBucket" => {
                    max_args(args, 1, method)?;
                    let bucket = match args.first() {
                        Some(Value::Str(b)) => b.clone(),
                        _ => self.bucket(&HashMap::new(), method)?,
                    };
                    let create = method == "createBucket";
                    thread_poller(
                        move || {
                            let body = if create && cfg.region != "us-east-1" && !cfg.custom_endpoint {
                                format!(
                                    "<CreateBucketConfiguration><LocationConstraint>{}</LocationConstraint></CreateBucketConfiguration>",
                                    cfg.region
                                )
                                .into_bytes()
                            } else {
                                Vec::new()
                            };
                            let m = if create { "PUT" } else { "DELETE" };
                            let r = request(&cfg, m, Some(&bucket), None, BTreeMap::new(), BTreeMap::new(), body)?;
                            ok_or(&format!("{} {}", if create { "createBucket" } else { "deleteBucket" }, bucket), r)?;
                            Ok(())
                        },
                        |_| Ok(Value::Nil),
                    )
                }
                "listBuckets" => {
                    max_args(args, 0, "listBuckets")?;
                    thread_poller(
                        move || {
                            let r = ok_or("listBuckets", request(&cfg, "GET", None, None, BTreeMap::new(), BTreeMap::new(), Vec::new())?)?;
                            let xml = String::from_utf8_lossy(&r.body).to_string();
                            let mut names: Vec<(String, String)> = blocks(&xml, "Bucket")
                                .iter()
                                .map(|b| (tag(b, "Name").unwrap_or_default(), tag(b, "CreationDate").unwrap_or_default()))
                                .collect();
                            names.sort();
                            Ok(names)
                        },
                        |names| {
                            Ok(list(
                                names
                                    .into_iter()
                                    .map(|(n, d)| map(vec![("name", Value::Str(n)), ("created", Value::Str(d))]))
                                    .collect(),
                            ))
                        },
                    )
                }
                _ => return Ok(None),
            }))
        })();
        match r {
            Ok(Some(p)) => Some(Ok(p)),
            Ok(None) => None,
            Err(e) => Some(Err(e)),
        }
    }

    fn methods(&self) -> Vec<&'static str> {
        vec![
            "put", "get", "getBytes", "download", "upload", "head", "exists", "delete", "list", "copy", "presign", "url",
            "withBucket", "createBucket", "deleteBucket", "listBuckets",
        ]
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    /// The example of the AWS documentation ("Signature Calculations for the
    /// Authorization Header: Transferring Payload in a Single Chunk", GET Object).
    #[test]
    fn aws_documentation_example() {
        let cfg = Cfg {
            endpoint: "https://s3.amazonaws.com".into(),
            custom_endpoint: false,
            region: "us-east-1".into(),
            access_key: "AKIAIOSFODNN7EXAMPLE".into(),
            secret_key: "wJalrXUtnFEMI/K7MDENG/bPxRfiCYEXAMPLEKEY".into(),
            session_token: None,
            bucket: Some("examplebucket".into()),
            path_style: false,
            timeout: Duration::from_secs(1),
        };
        let t = target(&cfg, Some("examplebucket"), Some("test.txt"));
        assert_eq!(t.host, "examplebucket.s3.amazonaws.com");
        let mut extra = BTreeMap::new();
        extra.insert("range".to_string(), "bytes=0-9".to_string());
        // 2013-05-24T00:00:00Z
        let h = sign(&cfg, "GET", &t, &BTreeMap::new(), &extra, &sha256_hex(b""), 1_369_353_600);
        assert_eq!(
            h["authorization"],
            "AWS4-HMAC-SHA256 Credential=AKIAIOSFODNN7EXAMPLE/20130524/us-east-1/s3/aws4_request, \
             SignedHeaders=host;range;x-amz-content-sha256;x-amz-date, \
             Signature=f0e8bdb87c964420e857bd35b5d6ed310bd44f0170aba48dd91039c6036bdb41"
        );
        // presigned URL example of the same documentation page set
        let url = presign(&cfg, "GET", "examplebucket", "test.txt", 86_400, 1_369_353_600);
        assert!(url.ends_with("X-Amz-Signature=aeeed9bbccd4d02ee5c0109b86d86835f995330da4c265957d157751f604d404"));
    }

    #[test]
    fn helpers() {
        assert_eq!(uri_encode("a b/ç+", true), "a%20b/%C3%A7%2B");
        assert_eq!(amz_dates(0).0, "19700101T000000Z");
        let cfg = parse_cfg(Some(&map(vec![
            ("endpoint", Value::Str("http://127.0.0.1:9000/".into())),
            ("accessKey", Value::Str("a".into())),
            ("secretKey", Value::Str("b".into())),
        ])))
        .unwrap();
        let t = target(&cfg, Some("bk"), Some("dir/x y.txt"));
        assert_eq!((t.host.as_str(), t.path.as_str()), ("127.0.0.1:9000", "/bk/dir/x%20y.txt"));
        assert_eq!(target(&cfg, Some("bk"), None).path, "/bk");
        let xml = "<a><Contents><Key>x&amp;y</Key><Size>3</Size></Contents><Contents><Key>z</Key></Contents></a>";
        assert_eq!(blocks(xml, "Contents").len(), 2);
        assert_eq!(tag(blocks(xml, "Contents")[0], "Key").unwrap(), "x&y");
    }
}
